#property copyright "TradeEdge AI Quantitative Systems"
#property version   "2.00"
#property strict

#include <Trade\Trade.mqh>
CTrade trade;

input double RiskPercent      = 1.0;      // Risk per trade (%)
input double MinRR            = 2.2;      // Minimum Reward:Risk Ratio
input int    FastEMA          = 50;       // HTF Trend Fast EMA
input int    SlowEMA          = 200;      // HTF Trend Slow EMA
input int    ADXPeriod        = 14;       // ADX Momentum Period
input double MinADX           = 20.0;     // Min ADX Trend Strength
input bool   MoveToBreakEven  = true;     // Auto Move SL to Break-Even at TP1

int handleEMA50, handleEMA200, handleADX, handleATR;
datetime lastBarTime = 0;

int OnInit() {
   handleEMA50  = iMA(_Symbol, _Period, FastEMA, 0, MODE_EMA, PRICE_CLOSE);
   handleEMA200 = iMA(_Symbol, _Period, SlowEMA, 0, MODE_EMA, PRICE_CLOSE);
   handleADX    = iADX(_Symbol, _Period, ADXPeriod);
   handleATR    = iATR(_Symbol, _Period, 14);
   return(INIT_SUCCEEDED);
}

void ManageBreakEven() {
   if(!MoveToBreakEven) return;
   for(int i = PositionsTotal() - 1; i >= 0; i--) {
      ulong ticket = PositionGetTicket(i);
      if(PositionSelectByTicket(ticket) && PositionGetString(POSITION_SYMBOL) == _Symbol) {
         double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
         double sl        = PositionGetDouble(POSITION_SL);
         double tp        = PositionGetDouble(POSITION_TP);
         double bid       = SymbolInfoDouble(_Symbol, SYMBOL_BID);
         double ask       = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
         long   type      = PositionGetInteger(POSITION_TYPE);

         if(type == POSITION_TYPE_BUY && sl < openPrice) {
            double halfWay = openPrice + (tp - openPrice) * 0.50;
            if(bid >= halfWay) trade.PositionModify(ticket, openPrice + (10 * _Point), tp);
         }
         if(type == POSITION_TYPE_SELL && (sl > openPrice || sl == 0)) {
            double halfWay = openPrice - (openPrice - tp) * 0.50;
            if(ask <= halfWay) trade.PositionModify(ticket, openPrice - (10 * _Point), tp);
         }
      }
   }
}

void OnTick() {
   ManageBreakEven();
   datetime currentBar = iTime(_Symbol, _Period, 0);
   if(currentBar == lastBarTime) return; // Closed-bar execution only
   lastBarTime = currentBar;

   if(PositionsTotal() > 0) return;

   double ema50[], ema200[], adx[], atr[];
   ArraySetAsSeries(ema50, true); ArraySetAsSeries(ema200, true);
   ArraySetAsSeries(adx, true);   ArraySetAsSeries(atr, true);

   if(CopyBuffer(handleEMA50, 0, 1, 2, ema50) < 2) return;
   if(CopyBuffer(handleEMA200, 0, 1, 2, ema200) < 2) return;
   if(CopyBuffer(handleADX, 0, 1, 2, adx) < 2) return;
   if(CopyBuffer(handleATR, 0, 1, 2, atr) < 2) return;

   double c1 = iClose(_Symbol, _Period, 1);
   double o1 = iOpen(_Symbol, _Period, 1);
   double h2 = iHigh(_Symbol, _Period, 2);
   double l2 = iLow(_Symbol, _Period, 2);
   double l0 = iLow(_Symbol, _Period, 1);
   double h0 = iHigh(_Symbol, _Period, 1);

   // Robot 1: FVG + Order Block Impulse
   bool bullFVG = (l0 > h2) && (c1 > o1);
   bool bearFVG = (h0 < l2) && (c1 < o1);

   // Robot 2: Trend + ADX Filter
   bool bullTrend = (ema50[0] > ema200[0]) && (adx[0] >= MinADX);
   bool bearTrend = (ema50[0] < ema200[0]) && (adx[0] >= MinADX);

   double slDistance = atr[0] * 1.5;
   double tpDistance = slDistance * MinRR;
   double balance    = AccountInfoDouble(ACCOUNT_BALANCE);
   double riskMoney  = balance * (RiskPercent / 100.0);
   double tickVal    = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   double tickSize   = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_SIZE);
   double lotStep    = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   double rawLots    = (slDistance > 0 && tickVal > 0) ? (riskMoney / ((slDistance / tickSize) * tickVal)) : 0.01;
   double lots       = MathMax(SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN), MathFloor(rawLots / lotStep) * lotStep);

   if(bullFVG && bullTrend) {
      double ask = SymbolInfoDouble(_Symbol, SYMBOL_ASK);
      trade.Buy(lots, _Symbol, ask, ask - slDistance, ask + tpDistance, "TradeEdge 4-Robot BUY");
   }
   else if(bearFVG && bearTrend) {
      double bid = SymbolInfoDouble(_Symbol, SYMBOL_BID);
      trade.Sell(lots, _Symbol, bid, bid + slDistance, bid - tpDistance, "TradeEdge 4-Robot SELL");
   }
}
